import React, { useEffect, useState } from 'react';
import { View, Text, TextInput, StyleSheet, ScrollView } from 'react-native';

export default function FormularioScreen({ route }) {

  // 📥 Datos que vienen desde la IA
  const { textoIA, datosIA } = route.params || {};

  // 🧠 Estados del formulario
  const [motivo, setMotivo] = useState('');
  const [observaciones, setObservaciones] = useState('');

  // 🤖 Auto-rellenar con datos de la IA
  useEffect(() => {
    if (datosIA) {
      setMotivo(datosIA.narrative || '');
      setObservaciones(datosIA.diagnosisPreliminary || '');
    }
  }, [datosIA]);

  return (
    <ScrollView contentContainerStyle={styles.container}>
      
      <Text style={styles.title}>Formulario Clínico</Text>

      {/* Motivo */}
      <Text style={styles.label}>Motivo de consulta</Text>
      <TextInput
        value={motivo}
        onChangeText={setMotivo}
        style={styles.input}
        placeholder="Motivo de consulta"
      />

      {/* Observaciones */}
      <Text style={styles.label}>Observaciones / Diagnóstico preliminar</Text>
      <TextInput
        value={observaciones}
        onChangeText={setObservaciones}
        style={styles.input}
        placeholder="Observaciones"
      />

      {/* Texto IA (debug útil) */}
      <Text style={styles.label}>Texto capturado por IA:</Text>
      <Text style={styles.textoIA}>
        {textoIA || 'No hay texto recibido'}
      </Text>

    </ScrollView>
  );
}

const styles = StyleSheet.create({
  container: {
    padding: 20,
  },
  title: {
    fontSize: 20,
    fontWeight: 'bold',
    marginBottom: 20,
  },
  label: {
    fontWeight: 'bold',
    marginTop: 10,
  },
  input: {
    borderWidth: 1,
    borderColor: '#ccc',
    padding: 10,
    borderRadius: 5,
    marginTop: 5,
  },
  textoIA: {
    marginTop: 10,
    fontStyle: 'italic',
  },
});
